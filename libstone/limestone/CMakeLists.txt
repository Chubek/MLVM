if(LIMESTONE_ENABLE_NATIVE_FFI)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LIMESTONE_LIBFFI QUIET IMPORTED_TARGET libffi)
  endif()
  if(TARGET PkgConfig::LIMESTONE_LIBFFI)
    target_link_libraries(exolayer PRIVATE $<BUILD_INTERFACE:PkgConfig::LIMESTONE_LIBFFI>)
  else()
    find_path(LIMESTONE_LIBFFI_INCLUDE_DIR ffi.h REQUIRED)
    find_library(LIMESTONE_LIBFFI_LIBRARY NAMES ffi REQUIRED)
    target_include_directories(exolayer SYSTEM PRIVATE ${LIMESTONE_LIBFFI_INCLUDE_DIR})
    target_link_libraries(exolayer PRIVATE $<BUILD_INTERFACE:${LIMESTONE_LIBFFI_LIBRARY}>)
  endif()
  target_compile_definitions(exolayer PRIVATE LIMESTONE_HAS_LIBFFI)
  target_link_libraries(exolayer PRIVATE $<INSTALL_INTERFACE:Limestone::native_ffi>)
endif()

cmake_minimum_required(VERSION 3.20)
project(Firestone VERSION 0.1.0 LANGUAGES CXX)

include(../cmake/LibstoneBuild.cmake)
include(CMakePackageConfigHelpers)
option(FIRESTONE_BUILD_TESTS "Build Firestone tests" "${LIBSTONE_BUILD_TESTS}")
if(FIRESTONE_BUILD_TESTS)
  enable_testing()
endif()

option(FIRESTONE_ENABLE_OPENMP "Enable OpenMP host compute parallelism" OFF)
find_package(Threads REQUIRED)
if(FIRESTONE_ENABLE_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
endif()

set(firestone_components ir rewrite eqsat cps memory object sema type vm bignum compute concurrency gfx intrin)
add_library(firestone INTERFACE)
foreach(component IN LISTS firestone_components)
  add_library(firestone_${component} INTERFACE)
  add_library(Firestone::${component} ALIAS firestone_${component})
  set_target_properties(firestone_${component} PROPERTIES EXPORT_NAME ${component})
  target_compile_features(firestone_${component} INTERFACE cxx_std_17)
  target_include_directories(firestone_${component} INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/firestone>)
endforeach()
add_library(Firestone::firestone ALIAS firestone)
target_link_libraries(firestone_eqsat INTERFACE firestone_rewrite)
target_link_libraries(firestone_cps INTERFACE firestone_ir)
target_link_libraries(firestone_type INTERFACE firestone_ir firestone_memory)
target_link_libraries(firestone_object INTERFACE firestone_ir firestone_type)
target_link_libraries(firestone_sema INTERFACE firestone_ir)
target_link_libraries(firestone_vm INTERFACE firestone_ir firestone_sema)
target_link_libraries(firestone_concurrency INTERFACE Threads::Threads)
target_link_libraries(firestone_compute INTERFACE Threads::Threads)
if(FIRESTONE_ENABLE_OPENMP)
  target_link_libraries(firestone_compute INTERFACE OpenMP::OpenMP_CXX)
endif()
foreach(component IN LISTS firestone_components)
  target_link_libraries(firestone INTERFACE firestone_${component})
  list(APPEND firestone_targets firestone_${component})
  list(APPEND firestone_directories lib${component})
  list(APPEND firestone_header_patterns lib${component}/*.hpp)
endforeach()

install(TARGETS firestone ${firestone_targets} EXPORT FirestoneTargets)
install(FILES Firestone.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/firestone)
install(DIRECTORY ${firestone_directories} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/firestone FILES_MATCHING PATTERN "*.hpp")
install(EXPORT FirestoneTargets NAMESPACE Firestone:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Firestone)
configure_package_config_file(cmake/FirestoneConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/FirestoneConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Firestone)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/FirestoneConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/FirestoneConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/FirestoneConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Firestone)

if(FIRESTONE_BUILD_TESTS)
  foreach(suite IN ITEMS ir representations rewrite eqsat memory types objects semantics cps algorithms extensions vm vm_compiler vm_binary vm_tiering bignum compute concurrency gfx intrin hardening)
    add_executable(firestone_${suite}_test tests/${suite}.cpp)
    target_link_libraries(firestone_${suite}_test PRIVATE Firestone::firestone)
    set_target_properties(firestone_${suite}_test PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(firestone_${suite}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME firestone.${suite} COMMAND firestone_${suite}_test)
    set_tests_properties(firestone.${suite} PROPERTIES TIMEOUT 60)
  endforeach()
  file(GLOB_RECURSE firestone_headers RELATIVE ${CMAKE_CURRENT_SOURCE_DIR} CONFIGURE_DEPENDS
    ${firestone_header_patterns})
  list(APPEND firestone_headers Firestone.hpp)
  foreach(header IN LISTS firestone_headers)
    string(MAKE_C_IDENTIFIER ${header} header_id)
    set(source ${CMAKE_CURRENT_BINARY_DIR}/headers/${header_id}.cpp)
    file(WRITE ${source} "#include \"${header}\"\n")
    list(APPEND header_sources ${source})
  endforeach()
  add_library(firestone_header_check OBJECT ${header_sources})
  target_link_libraries(firestone_header_check PRIVATE Firestone::firestone)
  set_target_properties(firestone_header_check PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(firestone_header_check PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME firestone.install COMMAND ${CMAKE_COMMAND}
    -DFIRESTONE_SOURCE=${CMAKE_CURRENT_SOURCE_DIR} -DFIRESTONE_BINARY=${CMAKE_CURRENT_BINARY_DIR}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/install.cmake)
endif()

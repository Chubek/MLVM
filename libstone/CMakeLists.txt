cmake_minimum_required(VERSION 3.20)
project(Libstone VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/LibstoneBuild.cmake)

if(LIBSTONE_BUILD_TESTS)
  enable_testing()
endif()
foreach(component IN ITEMS blackstone firestone limestone sandstone yellowstone)
  string(TOUPPER "${component}" key)
  option(LIBSTONE_BUILD_${key} "Build ${component}" ON)
  if(LIBSTONE_BUILD_${key})
    add_subdirectory(${component})
  endif()
endforeach()

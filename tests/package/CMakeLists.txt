cmake_minimum_required(VERSION 3.20)
project(MLVMConsumer LANGUAGES C CXX)
find_package(MLVM 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE MLVM::IR MLVM::Object)
if(TARGET MLVM::Binutils)
  target_link_libraries(consumer PRIVATE MLVM::Binutils)
  target_compile_definitions(consumer PRIVATE MLVM_TEST_BINUTILS=1)
endif()
if(TARGET MLVM::Cephyr)
  target_link_libraries(consumer PRIVATE MLVM::Cephyr)
  target_compile_definitions(consumer PRIVATE MLVM_TEST_CEPHYR=1)
  if(MLVM_CEPHYR_SYSROOT)
    target_compile_definitions(consumer PRIVATE MLVM_TEST_CEPHYR_SYSROOT="${MLVM_CEPHYR_SYSROOT}")
  endif()
endif()
add_executable(c-consumer ../core/c_api.c)
target_link_libraries(c-consumer PRIVATE MLVM::Core)
enable_testing()
add_test(NAME consumer COMMAND consumer)
add_test(NAME c-consumer COMMAND c-consumer)
